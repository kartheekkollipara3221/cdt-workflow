public override CAtlasApplicationConfig LoadStep(CAtlasPage input)
{
    string currentState = "Draft";
    string headerName = "Portfolio Trade Request";

    if (input != null && input.ID > 0)
    {
        CDTGeneral data = this.Context!.CDTGenerals.Single(x => x.Id == input.ID);

        if (!string.IsNullOrEmpty(data.WorkflowState))
            currentState = data.WorkflowState;

        if (!string.IsNullOrEmpty(data.RequestType))
            headerName = data.RequestType;

        if (!string.IsNullOrEmpty(data.AccountInformation))
            headerName = !string.IsNullOrEmpty(headerName)
                ? headerName + " - " + data.AccountInformation
                : data.AccountInformation;

        headerName = !string.IsNullOrEmpty(headerName)
            ? headerName + " - " + currentState
            : currentState;
    }

    return new CPAtlasApplicationConfig
    {
        Code = this.Code,
        Name = this.AppConfig!.Name,
        Steps = this.AppConfig!.Steps!.Select(x => new CApplicationStep
        {
            Active = x.Active,
            Disabled = x.Disabled,
            ID = x.ID,
            Label = x.Label,
            RouterLink = x.RouterLink,
            Status = x.Status,
            StepClass = x.StepClass,
            CurrentState = currentState,
            HeaderName = headerName
        })
    };
}

public override CApplicationStep LoadPage(CAtlasPage input)
{
    CApplicationStep step = new();

    if (this.AppConfig != null &&
        this.AppConfig.Steps != null &&
        this.AppConfig.Steps.Count > 0 &&
        input != null &&
        !string.IsNullOrEmpty(input.Step) &&
        this.AppConfig.Steps.Any(x =>
            string.Equals(x.ID, input.Step, StringComparison.InvariantCultureIgnoreCase)))
    {
        if (input.ID > 0)
        {
            step = LoadPageData(
                input,
                this.AppConfig.Steps.Single(x =>
                    string.Equals(x.ID, input.Step, StringComparison.InvariantCultureIgnoreCase)));
        }
        else
        {
            step = this.AppConfig.Steps.Single(x =>
                string.Equals(x.ID, input.Step, StringComparison.InvariantCultureIgnoreCase));
        }
    }

    if (string.IsNullOrEmpty(step.CurrentState))
        step.CurrentState = "Draft";

    if (string.IsNullOrEmpty(step.HeaderName))
        step.HeaderName = "Portfolio Trade Request";

    if (step.States!.Any(x =>
        string.Equals(x, step.CurrentState, StringComparison.InvariantCultureIgnoreCase)))
    {
        return step.DisableAllControls();
    }
    else
    {
        List<string> roles =
        [
            ..(from user in this.Context!.CPAtlasUsers
               join userRole in this.Context.CPAtlasUserEntitlements
                    on user.Id equals userRole.CpAtlasUserId
               join role in this.Context.CPAtlasEntitlements
                    on userRole.CpAtlasEntitlementId equals role.Id
               where user.LanId == this.User!.LanID
               select role.Name)
        ];

        if (step.Roles != null && step.Roles.Count > 0)
        {
            if (roles != null &&
                roles.Count > 0 &&
                roles.Intersect(step.Roles, StringComparer.InvariantCultureIgnoreCase).Any())
            {
                return step.DisableAllControls();
            }
        }
    }

    return step;
}

public override CAtlasPage SavePage(CAtlasPage input, IFormFileCollection files)
{
    CAtlasPage result = new();

    result.Code = input.Code;
    result.Step = input.Step;

    var transaction = this.Context!.Database.BeginTransaction();

    try
    {
        if (input.ID == 0)
        {
            Tuple<CDTGeneral, List<CPPageOutput>> output =
                MapData<CDTGeneral>(new CDTGeneral(), input.Outputs!);

            SetDefaultPageData(
                new CPAtlasPage
                {
                    Code = input.Code,
                    Step = input.Step
                },
                output.Item1);

            output.Item1.CreatedBy = this.User!.DisplayName;
            output.Item1.CreatedDate = DateTime.Now;

            this.Context.CDTGenerals.Add(output.Item1);
            this.Context.SaveChanges();
        }

this.Context.SaveChanges();

input.ID = output.Item1.Id;
result.ID = output.Item1.Id;
result.Outputs = output.Item2;

CDTGeneral general = this.Context.CDTGenerals.Single(x => x.Id == input.ID);

general.WorkflowState = this.Start(input, general);

if (string.Equals(general.WorkflowState, "Completed", StringComparison.InvariantCultureIgnoreCase))
    general.Status = "Trade Settled";
else
    general.Status = "Trade Request";

this.Context.SaveChanges();
}
else
{
    CDTGeneral general = this.Context.CDTGenerals.Single(x => x.Id == input.ID);

    Tuple<CDTGeneral, List<CPPageOutput>> output =
        MapData<CDTGeneral>(general, input.Outputs!);

    SetDefaultPageData(
        new CPAtlasPage
        {
            Code = input.Code,
            Step = input.Step
        },
        output.Item1);

    output.Item1.ModifiedBy = this.User!.DisplayName;
    output.Item1.ModifiedDate = DateTime.Now;

    this.Context.SaveChanges();

    result.ID = output.Item1.Id;
    result.Outputs = output.Item2;

    general.WorkflowState = this.Execute<CDTGeneral>(input, general);

    if (string.Equals(general.WorkflowState, "Completed", StringComparison.InvariantCultureIgnoreCase))
        general.Status = "Trade Settled";
    else
        general.Status = "Trade Request";

    this.Context.SaveChanges();
}

foreach (var file in files)
{
    if (file.Length > 0)
    {
        string folderPath = Path.Combine(
            AppDomain.CurrentDomain.BaseDirectory,
            this.Configuration!["AppSettings:CDTAttachmentFolderName"]!,
            new Random().Next().ToString());

        CDTAttachment attachment =
            MapData<CDTAttachment>(new CDTAttachment(), file, folderPath);

        attachment.CdtGeneralId = result.ID;
        attachment.CreatedBy = this.User!.DisplayName!;
        attachment.CreatedDate = DateTime.Now;

        this.Context.CDTAttachments.Add(attachment);

        result.Outputs.Add(new CPPageOutput
        {
            Name = attachment.Type,
            Value = attachment.FileName
        });
    }
}

if (input.Outputs!.Any(x =>
    string.Equals(x.Name, "SecurityIdentifier", StringComparison.InvariantCultureIgnoreCase)))
{
    CPPageOutput securityIdentifier =
        input.Outputs.Single(x =>
            string.Equals(x.Name, "SecurityIdentifier", StringComparison.InvariantCultureIgnoreCase));

    if (securityIdentifier != null && !string.IsNullOrEmpty(securityIdentifier.Value))
    {
        foreach (var data in JsonConvert.DeserializeObject<List<Dictionary<string, string>>>(securityIdentifier.Value)!)
        {
            if (ValidateData(data, ["Name", "Identifier", "Type", "Amount", "Side"]))
            {
                if (data.TryGetValue("Id", out string? value) &&
                    !string.IsNullOrEmpty(value) &&
                    Convert.ToInt32(value) > 0)
                {
                    CDTSecurityIdentifier security =
                        this.Context.CDTSecurityIdentifiers.Single(x => x.Id == Convert.ToInt32(value));

                    Tuple<CDTSecurityIdentifier, List<CPPageOutput>> output =
                        MapData<CDTSecurityIdentifier>(security, data);

                    output.Item1.CdtGeneralId = result.ID;
                    output.Item1.ModifiedBy = this.User!.DisplayName;
                    output.Item1.ModifiedDate = DateTime.Now;

                    result.Outputs.Add(new CPPageOutput
                    {
                        Name = "Security Identifier",
                        Value = JsonConvert.SerializeObject(output.Item2)
                    });
                }
                else
                {
                    Tuple<CDTSecurityIdentifier, List<CPPageOutput>> output =
                        MapData<CDTSecurityIdentifier>(new CDTSecurityIdentifier(), data);

                    output.Item1.CdtGeneralId = result.ID;
                    output.Item1.CreatedBy = this.User!.DisplayName;
                    output.Item1.CreatedDate = DateTime.Now;

                    this.Context.CDTSecurityIdentifiers.Add(output.Item1);

                    result.Outputs.Add(new CPPageOutput
                    {
                        Name = "Security Identifier",
                        Value = JsonConvert.SerializeObject(output.Item2)
                    });
                }
            }
        }
    }
}

this.Context.SaveChanges();
transaction.Commit();
}
catch (Exception ex)
{
    transaction.Rollback();
}

return result;

public void RemoveAttachment(int id)
{
    CDTAttachment attachment = this.Context!.CDTAttachments
        .Where(x => x.Id == id)
        .ToList()[0];

    attachment.IsDeleted = true;
    attachment.ModifiedBy = this.User!.DisplayName;
    attachment.ModifiedDate = DateTime.Now;

    this.Context.SaveChanges();
}

private CApplicationStep LoadPageData(CPAtlasPage input, CApplicationStep step)
{
    CDTGeneral data = this.Context!.CDTGenerals
        .ToList()
        .Where(x =>
            string.Equals(
                x.Id.ToString(),
                input.ID.ToString(),
                StringComparison.InvariantCultureIgnoreCase))
        .ToList()[0];

    LoadPageDataVisibility(step, data.WorkflowState!);

    if ((data.CdtAttachments == null ||
        (data.CdtAttachments != null && data.CdtAttachments.Count == 0)) &&
        this.Context.CDTAttachments.Any(x => x.CdtGeneralId == input.ID))
    {
        data.CdtAttachments =
        [
            ..this.Context.CDTAttachments.Where(x => x.CdtGeneralId == input.ID)
        ];
    }

    if (data.CdtAttachments != null && data.CdtAttachments.Count > 0)
    {
        foreach (CDTAttachment attachment in data.CdtAttachments.Where(x => x.IsDeleted == null || !x.IsDeleted.Value))
        {
            if (step.Controls!.Any(x =>
                string.Equals(x.Name, attachment.Type, StringComparison.InvariantCultureIgnoreCase)))
            {
                step.Controls!
                    .Single(x =>
                        string.Equals(x.Name, attachment.Type, StringComparison.InvariantCultureIgnoreCase))
                    .Files =
                    [
                        new()
                        {
                            ID = attachment.Id.ToString(),
                            Name = attachment.FileName,
                            URL = LoadAttachment(CPApplicationCode.CDT, attachment.ToCPAttachment())
                        }
                    ];

                step.Controls!
                    .Single(x =>
                        string.Equals(x.Name, attachment.Type, StringComparison.InvariantCultureIgnoreCase))
                    .Validator!.Required = false;

                step.Controls!
                    .Single(x =>
                        string.Equals(x.Name, attachment.Type, StringComparison.InvariantCultureIgnoreCase))
                    .Disabled = true;
            }
        }
    }

    if (step.Controls!.Any(x =>
        string.Equals(x.Name, "SecurityIdentifier", StringComparison.InvariantCultureIgnoreCase)))
    {
        if ((data.CdtSecurityIdentifiers == null ||
            (data.CdtSecurityIdentifiers != null && data.CdtSecurityIdentifiers.Count == 0)

&& this. Context. CDTSecurityIdentifiers.Any= x.CdtGeneralId == input.ID))

        {
            data.CdtSecurityIdentifiers =
            [
                ..this.Context.CDTSecurityIdentifiers.Where(x => x.CdtGeneralId == input.ID)
            ];
        }

        step.Controls!
            .Single(x =>
                string.Equals(x.Name, "SecurityIdentifier", StringComparison.InvariantCultureIgnoreCase));
    }
}
