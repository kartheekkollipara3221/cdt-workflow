namespace Secondarysales
{
    class BidsStatusChange
    {
        private static readonly ILogger log = LogManager.GetLogger(LogType.Log4Net, AppConfig.LogConfig);

        private const string constSitePath = "SitePath";
        private const string constBidslistkey = "List_SecondaryBids";
        private const string constOpportunitylistkey = "List_SecondaryOpportunities";
        private const string constDispositionEmailbody = "List_DispositionEmailBody";

        private const string constBidstatusAccepted = "ACCEPTED";
        private const string constBidstatusPartiallyAccepted = "PARTIALLY ACCEPTED";
        private const string constBidstatusNotAccepted = "NOT ACCEPTED";

        public bool BidssellerDisposition(AddV_InterfaceMaster oAddV_InterfaceMaster)
        {
            bool status;
            string Quarter;
            int daysToRespondBy;
            string queryBidsInfo;
            string Quarteryear;
            string vehicalName;
            string fundame;
            string accountNumber;
            string accountName;
            string amountCommited;
            string bidstatus;
            string BidsId;
            DateTime effectiveDateofSale;
            Addv EmailDetails oBidsEmailDetails;

Addv EmailDetails oBidsEmailDetails;
DataTable bidData;
SharePointDAL oSharePointDal;
ClientContext oClientContext;
List<Config> lSiteConfig;
List<Config> lBidsConfig;
List<Config> lAccpetedRejectedTextConfig;
XMLParser oXMLParser;
CamlQuery camlQueryBidsInfo;
SharePointPerson submitterPersonInfo;
EmailManager oEmailMgr;
EmailParameters oEmailParameters;

try
{
    status = true;
    oSharePointDal = new SharePointDAL();
    oXMLParser = new XMLParser();
    bidData = new DataTable();
    camlQueryBidsInfo = new CamlQuery();
    oEmailMgr = new EmailManager();
    oEmailParameters = new EmailParameters();

    BidsId = oAddV_InterfaceMaster.RunParameters.FirstOrDefault();

    lSiteConfig = oXMLParser.GetConfigDetails(
        oAddV_InterfaceMaster.InterfaceConfigPath,
        oAddV_InterfaceMaster.FeedName);

    oClientContext = oSharePointDal.CreateClientContext(
        lSiteConfig.FirstOrDefault(i => i.Key == constSitePath).Value,
        oAddV_InterfaceMaster.SPUserDomain,
        oAddV_InterfaceMaster.SPUserID,
        oAddV_InterfaceMaster.SPUserPWD);

    oBidsEmailDetails = oAddV_InterfaceMaster.AddV_EmailDetails
        .FirstOrDefault(i => i.EmailKey == oAddV_InterfaceMaster.FeedName);

    queryBidsInfo = oAddV_InterfaceMaster.SQLQueries
        .FirstOrDefault(i => i.Name == oAddV_InterfaceMaster.FeedName).SQL;

    camlQueryBidsInfo.ViewXml = queryBidsInfo.Replace("@BidsIDNumText", BidsId);

    lBidsConfig = oXMLParser.GetConfigDetails(
        oAddV_InterfaceMaster.InterfaceConfigPath,
        constBidslistkey);

    lAccpetedRejectedTextConfig = oXMLParser.GetConfigDetails(
        oAddV_InterfaceMaster.InterfaceConfigPath,
        constDispositionEmailbody);

    status = oSharePointDal.GetSharePointListData(
        oClientContext,
        lSiteConfig.FirstOrDefault(i => i.Key == constBidslistkey).Value,
        lBidsConfig,
        bidData,
        camlQueryBidsInfo);

    bidData.Columns.Add(
        lSiteConfig.FirstOrDefault(i => i.Key == "Quarter").Value);

bidData.Columns.Add(lSiteConfig.FirstOrDefault(i => i.Key == "Quarter").Value);
bidData.Columns.Add(lSiteConfig.FirstOrDefault(i => i.Key == "Quarter_Year").Value);

#region Classifying the data by Quarter and Quarter Year

foreach (DataRow opportunityDataRow in bidData.Rows)
{
    opportunityDataRow.BeginEdit();

    var effectiveDateOfSaleQuarter =
        Convert.ToDateTime(opportunityDataRow["Effective_Date_Of_Sale"]).getCurrentQuarter(true);

    opportunityDataRow["Quarter"] = effectiveDateOfSaleQuarter.Item1;
    opportunityDataRow["Quarter_Year"] = effectiveDateOfSaleQuarter.Item2.ToString();
}

bidData.AcceptChanges();

var grpOpportunityData =
    from DataRow OptDataRow in bidData.Rows
    orderby OptDataRow["Effective_Date_Of_Sale"]
    group OptDataRow by new
    {
        Quarter = OptDataRow["Quarter"],
        QuarterYear = OptDataRow["Quarter_Year"]
    };

#endregion

foreach (var OptDataRow in grpOpportunityData)
{
    Quarter = OptDataRow.FirstOrDefault()["Quarter"].ToString();
    QuarterYear = OptDataRow.FirstOrDefault()["Quarter_Year"].ToString();

    var toBuyerRMTeam =
        OptDataRow.FirstOrDefault().Field<string>("Buyer_RM_Team") == null
            ? string.Empty
            : String.Join(",",
                JsonConvert.DeserializeObject<List<SharePointPerson>>(
                    OptDataRow.FirstOrDefault().Field<string>("Buyer_RM_Team"))
                .Select(x => x == null ? string.Empty : x.Email));

    submitterPersonInfo =
        JsonConvert.DeserializeObject<List<SharePointPerson>>(
            OptDataRow.FirstOrDefault().Field<string>("Submitter"))
        .FirstOrDefault();

    vehicalName = OptDataRow.FirstOrDefault().Field<string>("Vehicle_Name_For_Opportunity");
    fundame = OptDataRow.FirstOrDefault().Field<string>("Opportunity_Fund");
    accountNumber = OptDataRow.FirstOrDefault().Field<string>("Bid_Submitted_Behalf_Account");

    accountName = OptDataRow.FirstOrDefault()
        .Field<string>("Bid_Submitted_Behalf_Account_Name")
        .Replace("\t", "")
        .Replace(" ", "");

    amountCommited = Decimal.Parse(
        OptDataRow.FirstOrDefault()
            .Field<string>("Commitment_Amount_To_Purchased")
            .Trim()).ToString("C");

    bidstatus = OptDataRow.FirstOrDefault().Field<string>("Bid_Status");

    effectiveDateOfSale = DateTime.Parse(
        OptDataRow.FirstOrDefault()
            .Field<string>("Effective_Date_Of_Sale"));

    oEmailParameters.Subject = oBidsEmailDetails.Subject
        .Replace("@Qtr", Quarter)
        .Replace("@YYYY", QuarterYear)
        .Replace("@VehicalName", vehicalName)
        .Replace("@AcctNum", accountNumber)
        .Replace("@AcctName", accountName);

oEmailParameters.Body = oBidsEmailDetails.Body
    .Replace("@AcctNum", accountNumber)
    .Replace("@AcctName", accountName)
    .Replace("@FundName", fundame)
    .Replace("@VehicalName", vehicalName)
    .Replace("@CommittedAmountForPurchase", amountCommited)
    .Replace("@BidID", BidsId);

switch (bidstatus.ToUpper())
{
    case constBidstatusAccepted:

        oEmailParameters.Body = oEmailParameters.Body
            .Replace("@AcceptedRejectedText",
                lAccpetedRejectedTextConfig.FirstOrDefault(i =>
                    i.Key.ToUpper() == constBidstatusAccepted.ToUpper()).Value)
            .Replace("@CurrentDate", effectiveDateOfSale.ToString("MM/dd/yyyy"));
        break;

    case constBidstatusPartiallyAccepted:

        oEmailParameters.Body = oEmailParameters.Body
            .Replace("@AcceptedRejectedText",
                lAccpetedRejectedTextConfig.FirstOrDefault(i =>
                    i.Key.ToUpper() == constBidstatusPartiallyAccepted.ToUpper()).Value)
            .Replace("@CurrentDate", effectiveDateOfSale.ToString("MM/dd/yyyy"));
        break;

    case constBidstatusNotAccepted:

        oEmailParameters.Body = oEmailParameters.Body
            .Replace("@AcceptedRejectedText",
                lAccpetedRejectedTextConfig.FirstOrDefault(i =>
                    i.Key.ToUpper() == constBidstatusNotAccepted.ToUpper()).Value);
        break;

    default:

        log.Debug(string.Format(
            "Bid Disposition received with an unknown status : \"{0}\" ",
            bidstatus));
        break;
}

oEmailParameters.CC = oBidsEmailDetails.CCEmail;
oEmailParameters.BCC = oBidsEmailDetails.BCCEmail;
oEmailParameters.From = oBidsEmailDetails.FromEmail;

// PBS-897
// Temporarily removed Bid Submitted By from
// "To" email and sent to "CC" email address

oEmailParameters.To = toBuyerRMTeam + ", " + submitterPersonInfo.Email;
oEmailParameters.Host = AppConfig.SMTPHost;
oEmailParameters.Port = AppConfig.SMTPPort;

oEmailMgr.SendEmail(oEmailParameters);
}
return status;
}
finally
{
}
}

public bool BidssubmittedActions(AddV_InterfaceMaster oAddV_InterfaceMaster)
{
    XMLParser oXmlParser;
    SharePointDAL oSharePointDAL;
    ClientContext oClientContext;
    List<Config> lConfig;
    List<Config> lBidsFieldsConfig;
    List<Config> lOpportunitysFieldsConfig;
    CamlQuery submittedBidCamlQuery;
    CamlQuery bidsofopportunityCamlQuery;
    CamlQuery opportunityCamlQuery;
    DataTable submittedBidData;
    DataTable bidsofopportunityData;
    DataTable opportunityData;
    string bidIDArgument;
    string querysubmittedBid;
    string queryBidsofopportunity;
    string queryopportunity;
    string opportunityID;
    string maxAmountValueString;
    string minAmountValueString;
    string maxPercentFairValueString;
    string minPercentFairValueString;
    string countNumberofBids;
    bool status;

    try
    {
        oXmlParser = new XMLParser();

        lConfig = oXmlParser.GetConfigDetails(
            oAddV_InterfaceMaster.InterfaceConfigPath,
            oAddV_InterfaceMaster.FeedName);

        oSharePointDAL = new SharePointDAL();

        oClientContext = oSharePointDAL.CreateClientContext(
            lConfig.FirstOrDefault(i => i.Key == constSitePath).Value,
            oAddV_InterfaceMaster.SPUserDomain,
            oAddV_InterfaceMaster.SPUserID,
            oAddV_InterfaceMaster.SPUserPWD);

        bidIDArgument = oAddV_InterfaceMaster.RunParameters.FirstOrDefault();

        // Get the Opportunity ID Information from BID

        submittedBidData = new DataTable();
        submittedBidCamlQuery = new CamlQuery();

submittedBidCamlQuery = new CamlQuery();

lBidsFieldsConfig = oXmlParser.GetConfigDetails(
    oAddV_InterfaceMaster.InterfaceConfigPath,
    constBidslistkey);

querySubmittedBid = oAddV_InterfaceMaster.SQLQueries
    .FirstOrDefault(i => i.Name == "BidsRequesterInfo").Sql;

submittedBidCamlQuery.ViewXml =
    querySubmittedBid.Replace("@BidsIDNumText", bidIDArgument);

status = oSharePointDAL.GetSharePointListData(
    oClientContext,
    lConfig.FirstOrDefault(i => i.Key == constBidslistkey).Value,
    lBidsFieldsConfig,
    submittedBidData,
    submittedBidCamlQuery);

if ((submittedBidData == null) || (submittedBidData.Rows.Count != 1))
{
    throw new Exception("Issue Retriving Data of Submitted Bid or More than one Bid Information retrived");
}
// Get List Of Bids for the Opportunity
// for which a Bid has been submitted

opportunityID = submittedBidData.Rows[0]["Opportunity_ID"].ToString();

bidsofopportunityData = new DataTable();
bidsofopportunityCamlQuery = new CamlQuery();

lBidsFieldsConfig = oXmlParser.GetConfigDetails(
    oAddV_InterfaceMaster.InterfaceConfigPath,
    constBidslistkey);

queryBidsofopportunity = oAddV_InterfaceMaster.SQLQueries
    .FirstOrDefault(i => i.Name == "Bidsofopportunity").Sql;

bidsofopportunityCamlQuery.ViewXml =
    queryBidsofopportunity.Replace("@OpportunityIDNumText", opportunityID);

status = oSharePointDAL.GetSharePointListData(
    oClientContext,
    lConfig.FirstOrDefault(i => i.Key == constBidslistkey).Value,
    lBidsFieldsConfig,
    bidsofopportunityData,
    bidsofopportunityCamlQuery);

if ((bidsofopportunityData == null) || (bidsofopportunityData.Rows.Count < 1))
{
    throw new Exception("Issue Retriving Bids of a Opportunity of Submitted Bid");
}
// Calculate the Bid counts, Max and Min Bid Amount

countNumberOfBids = bidsofopportunityData.Rows.Count.ToString();

maxAmountValueString = GetAmountValueFromTheBidsList(bidsofopportunityData, true);
minAmountValueString = GetAmountValueFromTheBidsList(bidsofopportunityData, false);

maxPercentFairValueString =
    GetPercentFairValueFromTheBidsList(bidsofopportunityData, true);

minPercentFairValueString =
    GetPercentFairValueFromTheBidsList(bidsofopportunityData, false);

// Get The Opportunity Information and Update the Opportunity with latest data

opportunityData = new DataTable();
opportunityCamlQuery = new CamlQuery();

lOpportunitysFieldsConfig = oXmlParser.GetConfigDetails(
    oAddV_InterfaceMaster.InterfaceConfigPath,
    constOpportunitylistkey);

queryOpportunity = oAddV_InterfaceMaster.SQLQueries
    .FirstOrDefault(i => i.Name == "OpportunityRequesterInfo").Sql;

opportunityCamlQuery.ViewXml =
    queryOpportunity.Replace("@OpportunityIDNumText", opportunityID);

status = oSharePointDAL.GetSharePointListData(
    oClientContext,
    lConfig.FirstOrDefault(i => i.Key == constOpportunitylistkey).Value,
    lOpportunitysFieldsConfig,
    opportunityData,
    opportunityCamlQuery);

if ((opportunityData == null) || (opportunityData.Rows.Count != 1))
    throw new Exception("Issue Retriving Opportunity Data of Submitted Bid");

// Set & Update Data for the Opportunity

opportunityData.Rows[0].SetField<string>("BidCount", countNumberOfBids);
opportunityData.Rows[0].SetField<string>("BidMaxValue", maxAmountValueString);
opportunityData.Rows[0].SetField<string>("BidMinValue", minAmountValueString);
opportunityData.Rows[0].SetField<string>("Lowest_Bid_Percent_Of_FairValue", minPercentFairValueString);
opportunityData.Rows[0].SetField<string>("Largest_Bid_Percent_Of_FairValue", maxPercentFairValueString);

status = oSharePointDAL.UpdateDataInSharePointList(
    oClientContext,
    lConfig.FirstOrDefault(i => i.Key == constOpportunitylistkey).Value,
    "SP_ID",
    lOpportunitysFieldsConfig,
    opportunityData);

if (!status)
{
    throw new Exception("Issue Updating Opportunity Data of Submitted Bid");
}
return status;
}
finally
{
}
}
private string GetAmountValueFromTheBidsList(DataTable inputBidsData, bool getMaxValue)
{
    List<double> outputData;

    try
{

if (getMaxValue)
{
    // Get Max Value from Bids of the Opportunity Data
    outputData = (
        from DataRow bidDataRow in inputBidsData.AsEnumerable()
        group bidDataRow by bidDataRow.Field<string>("Opportunity_ID") into groupedData
        select groupedData.Max(i =>
            double.Parse(i.Field<string>("Commitment_Amount_To_Purchased")))
    ).ToList();
}
else
{
    // Get Min Value from Bids of the Opportunity Data
    outputData = (
        from DataRow bidDataRow in inputBidsData.AsEnumerable()
        group bidDataRow by bidDataRow.Field<string>("Opportunity_ID") into groupedData
        select groupedData.Min(i =>
            double.Parse(i.Field<string>("Commitment_Amount_To_Purchased")))
    ).ToList();
}

return outputData.FirstOrDefault().ToString();
}
finally
{
}

private string GetPercentFairValueFromTheBidsList(DataTable inputBidsData, bool getMaxValue)
{
    List<double> outputData;

    try
    {
        if (getMaxValue)
        {
            // Get Max Value from Bids of the Opportunity Data
            outputData = (
                from DataRow bidDataRow in inputBidsData.AsEnumerable()
                group bidDataRow by bidDataRow.Field<string>("Opportunity_ID") into groupedData
                select groupedData.Max(i =>
                    double.Parse(i.Field<string>("Bid_Expressed_As_Fair_Value")))
            ).ToList();
        }
        else
{

        {
            // Get Min Value from Bids of the Opportunity Data
            outputData = (
                from DataRow bidDataRow in inputBidsData.AsEnumerable()
                group bidDataRow by bidDataRow.Field<string>("Opportunity_ID") into groupedData
                select groupedData.Min(i =>
                    double.Parse(i.Field<string>("Bid_Expressed_As_Fair_Value")))
            ).ToList();
        }

        return outputData.FirstOrDefault().ToString();
    }
    finally
    {
    }
}
}
