<SqlQueryList>

    <SqlQuery Name="SecondarySalesUploadAccountData">
        <Sql>
            SELECT
                ACCOUNT.ACCOUNT_NUMBER AS Account_Number,
                ACCOUNT.B12_LONG_NAME AS Account_Name,
                ACCOUNT.B171_ADMIN_OFFICER_NAME AS RM_Name,
                ACCOUNTUSER.UD2_EMPLOYEE_TYPE AS Account_Level_Coding,
                ACCOUNT.B175_BRANCH_NUMBER_NAME AS AccountBranchOffice
            FROM
                ADDV_ACCOUNT ACCOUNT,
                ADDV_ACCOUNTUSER ACCOUNTUSER
            WHERE ACCOUNT.ACCOUNT_NUMBER = ACCOUNTUSER.ACCOUNT_NUMBER AND
                ACCOUNT.P18_DATE_CLOSED IS NULL AND
                ACCOUNTUSER.UD2_EMPLOYEE_TYPE IS NOT NULL
        </Sql>
    </SqlQuery>
