<SqlQueryList>

    <SqlQuery Name="AccountPopulation">
        <Sql>
            SELECT
                DIM_ACCOUNT.ACCOUNT_NUMBER AS ACCOUNT_NUMBER,
                DIM_ACCOUNT.ACCOUNT_SHORT_NAME AS ACCOUNT_SHORT_NAME,
                DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE AS ACCOUNT_SUB_TYPE_CODE,
                REF_CODE_DESC.DESCRIPTION AS ACCOUNT_SUB_TYPE_DESCRIPTION,
                VW_DIM_ACCOUNT_UD.UD_45 AS UD_45
            FROM
                PBDWAPP.DIM_ACCOUNT DIM_ACCOUNT
                LEFT OUTER JOIN
                PBDWAPP.VW_DIM_ACCOUNT_UD VW_DIM_ACCOUNT_UD
                    ON (DIM_ACCOUNT.ACCOUNT_NUMBER = VW_DIM_ACCOUNT_UD.ACCOUNT_NUMBER)
                LEFT OUTER JOIN
                PBDWAPP.REF_CODE_DESC REF_CODE_DESC
                    ON (DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE = PBDWAPP.REF_CODE_DESC.CODE
                    AND REF_CODE_DESC.TABLE_NUMBER = '22')
            WHERE
                DIM_ACCOUNT.END_DATE IS NULL AND
                DIM_ACCOUNT.DATE_CLOSED IS NULL AND
                DIM_ACCOUNT.ADMIN_BRANCH_NUMBER IN (1,2,3,4,6,8,10,21,22)

            UNION

            SELECT
                DIM_ACCOUNT.ACCOUNT_NUMBER AS ACCOUNT_NUMBER,
                DIM_ACCOUNT.ACCOUNT_SHORT_NAME AS ACCOUNT_SHORT_NAME,
                DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE AS ACCOUNT_SUB_TYPE_CODE,
                REF_CODE_DESC.DESCRIPTION AS ACCOUNT_SUB_TYPE_DESCRIPTION,
                VW_DIM_ACCOUNT_UD.UD_45 AS UD_45
            FROM
                PBDWAPP.DIM_ACCOUNT DIM_ACCOUNT
                LEFT OUTER JOIN
                PBDWAPP.VW_DIM_ACCOUNT_UD VW_DIM_ACCOUNT_UD
                    ON (DIM_ACCOUNT.ACCOUNT_NUMBER = VW_DIM_ACCOUNT_UD.ACCOUNT_NUMBER)
                LEFT OUTER JOIN
                PBDWAPP.REF_CODE_DESC REF_CODE_DESC
                    ON (DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE = PBDWAPP.REF_CODE_DESC.CODE
                    AND REF_CODE_DESC.TABLE_NUMBER = '22')
            WHERE
                DIM_ACCOUNT.END_DATE IS NULL AND
                DIM_ACCOUNT.DATE_CLOSED IS NOT NULL AND
                DIM_ACCOUNT.ADMIN_BRANCH_NUMBER IN (1,2,3,4,6,8,10,21,22)

            UNION

            SELECT
                DIM_ACCOUNT.ACCOUNT_NUMBER AS ACCOUNT_NUMBER,
                DIM_ACCOUNT.ACCOUNT_SHORT_NAME AS ACCOUNT_SHORT_NAME,
                DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE AS ACCOUNT_SUB_TYPE_CODE,
                REF_CODE_DESC.DESCRIPTION AS ACCOUNT_SUB_TYPE_DESCRIPTION,
                VW_DIM_ACCOUNT_UD.UD_45 AS UD_45
            FROM
                PBDWAPP.DIM_ACCOUNT DIM_ACCOUNT
                LEFT OUTER JOIN
                PBDWAPP.VW_DIM_ACCOUNT_UD VW_DIM_ACCOUNT_UD
                    ON (DIM_ACCOUNT.ACCOUNT_NUMBER = VW_DIM_ACCOUNT_UD.ACCOUNT_NUMBER)
                LEFT OUTER JOIN
                PBDWAPP.REF_CODE_DESC REF_CODE_DESC
                    ON (DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE = PBDWAPP.REF_CODE_DESC.CODE
                    AND REF_CODE_DESC.TABLE_NUMBER = '22')
            WHERE
                DIM_ACCOUNT.END_DATE IS NULL AND
                DIM_ACCOUNT.DATE_CLOSED IS NULL AND
                DIM_ACCOUNT.ADMIN_BRANCH_NUMBER IS NULL AND
                DIM_ACCOUNT.ACCOUNT_SUB_TYPE_CODE IN ('122')
        </Sql>
    </SqlQuery>

</SqlQueryList>
