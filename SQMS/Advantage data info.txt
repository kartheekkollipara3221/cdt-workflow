using System.Collections.Generic;
using System.Ling;
using System.Text;
using System.Threading.Tasks;
using System.Data;
using PB.DataAccess;
using PB.CommonLib;
using PB.Logger;
using Microsoft.SharePoint.Client;
using system;

namespace SecondarySales
{
    class AddvantageDataInfo
    {
        #region Global variables

        private static Ilogger log = LogManager.GetLogger(LogType.Log4Net, AppConfig.LogConfig);

        private const string constAddVantageInfolistkey = "List_AddvAccountInfo";
        private const string constAccountNumberCol = "Account_Number";

        #endregion

        #region Public methods

        public bool UploadAccount(Addv_InterfaceMaster OAddv_InterfaceMaster)
        {
            SharePointDAL oSharePointDAL;
            ClientContext oClientContext;
            XMLParser oXmlParser;
            SgIDAL OSQLDAL;
            AddV_InterfaceDetails oAddVInterfaceDeatils;
            DataManager oDataManager;
            List<Config> IsSiteConfig;
            List<Config> lsAddVantageInfoConfig;
            List<Config> IsUploadAddVantageInfoConfig;
            List<string> IsSkipColumns;
            Datalable SharePointData;
            DataTable ODWHData;
            DataTable outputNewData;
            DataTable outputUpdatedData;
            string sqlQuery;
            string sitePath;
            string addvantageInfoListName;
            bool status;

try
{
    log.MethodStart();

    oXmlParser = new XMLParser();
    oSharePointDAL = new SharePointDAL();
    oSharePointData = new DataTable();
    OSQLDAL = new SqlDAL();
    ODataManager = new DataManager();
    lsSkipColumns = new List<string>();

    oAddVInterfaceDeatils = oAddV_InterfaceMaster.AddVInterfaceDetails.FirstOrDefault(
        i => i.InterfaceKey == oAddV_InterfaceMaster.FeedName
    );

    lsSiteConfig = oXmlParser.GetConfigDetails(
        oAddV_InterfaceMaster.InterfaceConfigPath,
        oAddV_InterfaceMaster.FeedName
    );

    lsAddVantageInfoConfig = oXmlParser.GetConfigDetails(
        oAddV_InterfaceMaster.InterfaceConfigPath,
        constAddVantageInfoListKey
    );

    sqlQuery = oAddV_InterfaceMaster.SQLQueries
        .FirstOrDefault(i => i.Name == oAddV_InterfaceMaster.FeedName).Sql;

    sitePath = lsSiteConfig
        .FirstOrDefault(i => i.Key == "SitePath").Value;

    addVantageInfoListName = lsSiteConfig
        .FirstOrDefault(i => i.Key == constAddVantageInfoListKey).Value;

    oClientContext = oSharePointDAL.CreateClientContext(
        sitePath,
        oAddV_InterfaceMaster.SPUserDomain,
        oAddV_InterfaceMaster.SPUserID,
        oAddV_InterfaceMaster.SPUserPWD
    );

    log.Debug("Get Data From SharePoint list");

    status = oSharePointDAL.GetSharePointListData(
        oClientContext,
        addVantageInfoListName,
        lsAddVantageInfoConfig,
        oSharePointData
    );

    //status = oSharePointDAL.DeleteDataInSharePointList(
    //    oClientContext,
    //    addVantageInfoListName,
    //    "SP_ID",
    //    oSharePointData
    //);

    log.Debug("Get Data From ADDV");

    oDWHData = OSQLDAL.ExecuteQueryODBC(
        oAddV_InterfaceMaster.AddVantageDBConnection,
        sqlQuery
    );

    outputNewData = oDataManager.CompareDataTablesGetNewData(
        oDWHData,
        oSharePointData,
        constAccountNumberCol
    );

    lsSkipColumns.Add("Date Closed");
    lsSkipColumns.Add("SP ID");

    outputUpdatedData = oDataManager.CompareDataTablesGetChangedData(
        oDWHData,
        oSharePointData,
        lsSkipColumns,
        constAccountNumberCol,
        false
    );

    lsUploadAddVantageInfoConfig = lsAddVantageInfoConfig
        .Where(i => i.Key != "ID")
        .ToList();

    lsUploadAddVantageInfoConfig = lsAddVantageInfoConfig
        .Where(i => i.Key != "ID")
        .ToList();

status = oSharePointDAL.ExportDataToSharePointList(
    oClientContext,
    addvantageInfoListName,
    lsUploadAddVantageInfoConfig,
    outputNewData
);

status = oSharePointDAL.UpdateDataInSharePointList(
    oClientContext,
    addvantageInfoListName,
    "SP_ID",
    lsAddVantageInfoConfig,
    outputUpdatedData
);

return status;
}
finally
{
}
